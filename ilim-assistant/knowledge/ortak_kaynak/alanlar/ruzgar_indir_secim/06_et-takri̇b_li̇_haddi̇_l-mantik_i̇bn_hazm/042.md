النفــوس. ][[1 [ واعلــم أن الســور الكلّــي لا يجــوز أن يوضــع إلا قبــل الموضــوع أي قبــل المخبــر عنــه لا قبــل المحمــول، وهــو الخبــر، لأنــك إذا قلــت: كلّ إنســان حــي صدقــت، وإذا قلــت: الإنســان كلّ حــيّ، كذبــت؛ وإنمــا يجــوز أن تقــرن الســور الكلّــي بالمحمــول إذا كان حــدًّا أو رســمًا، كقولــك: نفــس الإنســان كلّ حيــة ناطقــة مبتــة ذات جســد ملــوّن، وإذا قلــت: الإنســان كلّ ضحّــاك، أو إذا كان الســور جزئيًــا مثــل قولــك: الإنســان بعــض الحيّ.
