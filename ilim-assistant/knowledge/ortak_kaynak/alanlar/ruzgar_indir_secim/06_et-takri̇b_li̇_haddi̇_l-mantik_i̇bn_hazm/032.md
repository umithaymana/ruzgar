لمــن كابــر حسّــه وناكــر عقلــه، نعــوذ بــاللّ مــن ذلــك. ]266 [ والوجــه الثانــي أن الــذي لــم يحقّــق النظــر لمــا لــم يقــدر علــى إمســاك الزمــان وقتيــن تفلــت عليــه ضبــط الزمــان المقيــم ولــم يكــد يتحقّــق ذلــك لحسّــه. ]26[ [ فليعلــم أن الزمــان لا يثبــت وإنمــا هــو منقــض أبــدًا شــيئًا بعد شــيء، والزمان هو فصل موجود أبدًا بين الزمان الماضي والزمان »الآن« المقيم هو الآن؛ فإن قولك الآتــي؛ والآن هــو الموجــود فــي الحقيقــة مــن الزمــان أبدًا، وما قبل الآن فماض وما بعد الآن فمستقبل؛ إلا أن العبارة في اللغة العربية عن الزمان المقيم والزمان المستقبل بلفظ واحــد وهــو فــي اللغــة الأعجميــة بصيغتيــن مختلفتين وذلك أوضح في البيان والإفهام.
