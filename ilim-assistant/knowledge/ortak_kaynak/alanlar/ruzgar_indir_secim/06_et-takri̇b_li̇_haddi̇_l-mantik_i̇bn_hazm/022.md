التــي تعجّبــت منهــا. ]149 [ وقــد ظــنّ قــوم أن مــن الخبــر مــا لا يدخلــه صــدق ولا كــذب، وهــو إخبــار الهــاذي بأمــر متيقّــن صحّتــه، لــم يقصــده، كقولــه: لا إلــه إلا اللّ، مــات الرجــل فهــذان خبــران صحيحــان. ]150 [ قــال أبــو محمّــد: وهــذا الظــنّ غيــر صحيــح لأنــه لا يكــون مخبــرًا إلا مــن قصــد الخبــر، والخبــر مــع المخبــر مــن بــاب الإضافــة، فــا خبــر إلا لمخبــر، والهــاذي ليــس مخبــرًا بكامــه، فكامــه ليــس خبــرًا. ]151 [ وإذ قــد وافانــا ذكــر الأمــر فلنقــل علــى أقســامه قــولاً وجيزًا؛ وذلــك أن الأمر ينقســم أقســامًا منهــا: الواجــب الملــزم، وهــو عنصــر الأمــر الــذي لا ينتقــل عنــه لفــظ الأمــر إلا بدليــل برهانــي، ومنهــا المحضــوض عليــه غيــر الملــزم، ومنهــا المبــاح، ومنهــا المســموح فيــه، وهــو الــذي تركــه أفضــل؛ ومنهــا التبــرؤ كقــول القائــل: اعمــل مــا شــئت؛
