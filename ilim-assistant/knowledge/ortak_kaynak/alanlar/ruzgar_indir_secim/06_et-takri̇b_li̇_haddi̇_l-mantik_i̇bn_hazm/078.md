وجــود مــن نمنــع أن ــاً أيض يجــب ، النحــاس يجــذب حجــر، ولابــد، يوجــد لا أنــه علــى نقطــع أن يجــب فليــس الحديــد .كثيــر ذلــك ومثــل. يوجــد أنــه علــى ــاً أيض نقطــع أن ولا ][1[ [ ــلّوحل هــذا مّحــر قــد بأنــه لّالأو الخالــق علــى التحكــم ــهّكل هــذا مــنّ وأشــد آخــر اًشــيئ مّحــر قــد لأنــه، النفــوس بشــهوات لكــن، بذلــك تعالــى عنــه وارد حكــم بــا، هــذا عصــاة ــاًأناس تعالــى ّالل لعــن ولقــد، بلــى. صفاتــه بعــض فــي أنــت مّتحــر الــذي هــذا يشــبه عصاة وأنهم ناس أنهــم في الملعونين أولئــك مثــل لأنهــم إنســانّ كل يلعــن افتــراه، يــنّمعتد ّ كل افترى ونســاءهم أطفالهم معهم وأحرق، المكيال خانوا لأنهم اًقوم وأحرق ونّمعتد .البعيــد الضــال لهــو هــذا إن وامرأتــه وولــده هــو يحــرق والميــزان للمكيــال خائــن ][19 [ يبــدو أن ــواّأحب إذا أنهــم الفاســد النــوع بهــذا للمغروريــن يقــع مــا بديــع ومــن أشــياء فيخرجــون، تركــه عليهــم يصعــب لــم بــه حكمــوا الــذي الحكــم هــذا فــي لهــم ّ، وشــذ أصلــه عــن خــرج هــذا: ويقولــون بعضهــا فــي وجــد حكــم عــن المشــتبهات مــن للشــاذ ًأصــا عنــه المشــذوذ الحكــم هــذا كان لــو: نقــول ونحــن. عليــه يقــاس لاّ والشــاذ لمــا، ذلــك كان ولــو. أصلــه غيــر مــن فــرع ينبعــث أن يجــوز ولاّ، شــذ مــا عنــهّ شــذ لمــا ولا. منــه ًمتأصــا الأصــل مــن المتأصــل كان ولا، بــه للمتأصــل ًأصــا الأصــل كان .فتخطــئ وأصلــه نوعــه عــنّ شــاذ أنــه لــه الجامعــة
