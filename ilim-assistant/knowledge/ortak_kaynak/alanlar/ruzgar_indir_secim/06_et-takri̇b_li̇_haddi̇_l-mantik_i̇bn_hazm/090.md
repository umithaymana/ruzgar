تتــردّد، ولا تــرض لنفســك ببقــاء ســاعة آبيــا مــن قبــول الحــقّ. وإن وجــدت تمويهــا فبينــه ولا تغتــر بذهــاب خصمــك عنــه فلعــلّ غيــره مــن أهــل مقالتــه يتفطــن لمــا غــاب عنــه. ][05 [ فلــك، مبطــل إنــك عنــك قيــل أن تبــال ولا الظفــر بحقيقــة إلا تقنــع ولا هــذا فمــن الســام عليهــم الأنبيــاء مــن أســوة أكــرم قيــنّالمحق مــن ذلــك إليــه نســب فيمــن تســتوحش ولا. إليــه للباطــل ونســبة لحقــه ــاًدفع قتــل منهــم اً كثيــر إن حتــى نعــم. دونهــم بكثــرة تبــال ولا. معــه تعالــى فالخالــقّ الحــق معــه كان فمــن، أحــد إلــىّ الحــق مــع منهــم أكثــرّ فالحــق تهــمّبعز ولا إياهــم النــاس بتعظيــم ولا أزمانهــم بقــدم ولا خصومــك .بالتعظيــم وأولــى أحــدّ كل عنــد وأعظــمّ وأعــز وأقــدم ][06 [ ،ــةّأمّ وكل ــةّملّ كل أهــل ــلّفتأم ذكرنــا مــا مراعــاة فســاد ــنّتتيق أن شــئت وإذا ّ وذم خيــرّ وبــكل فضيلــةّ بــكل وصفتهــم أســافهم تعظيــم علــى مطبقيــن تجدهــم فإنــك ،ً قليا أمــره لّ أو فــي بــه القائلــون كان فقــد، يقــال قــولّ كل ــلّوتأم.
